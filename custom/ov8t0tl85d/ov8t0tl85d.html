CustomFishingRodItem.java

// src/main/java/net/example/fishingmod/item/CustomFishingRodItem.java
package net.example.fishingmod.item;

import net.example.fishingmod.FishingMod;
import net.example.fishingmod.screen.FishingHudScreen;
import net.minecraft.client.MinecraftClient;
import net.minecraft.entity.player.PlayerEntity;
import net.minecraft.item.FishingRodItem;
import net.minecraft.item.ItemStack;
import net.minecraft.sound.SoundEvents;
import net.minecraft.stat.Stats;
import net.minecraft.util.Hand;
import net.minecraft.util.TypedActionResult;
import net.minecraft.world.World;

public class CustomFishingRodItem extends FishingRodItem \{
public CustomFishingRodItem(Settings settings) \{
super(settings);
\}

@Override
public TypedActionResult < ItemStack > use(World world, PlayerEntity user, Hand hand) \{
ItemStack itemStack = user.getStackInHand(hand);

if (user.fishHook != null) \{
// Recall fishing bobber
user.fishHook.onDestroyedBy(user);
world.playSoundFromEntity(null, user, SoundEvents.ENTITY_FISHING_BOBBER_RETRIEVE, 1.0F, 1.0F);
\} else \{
// Open custom HUD instead of casting
if (!world.isClient) \{
// Server-side: sync to client
return TypedActionResult.success(itemStack);
\} else \{
MinecraftClient.getInstance().setScreen(new FishingHudScreen());
world.playSound(user, user.getX(), user.getY(), user.getZ(),
SoundEvents.ENTITY_FISHING_BOBBER_THROW, 1.0F, 1.0F);
\}
\}

user.incrementStat(Stats.USED.getOrCreateStat(this));
return TypedActionResult.success(itemStack);
\}
\}

// src/main/java/net/example/fishingmod/block/FishBlock.java
package net.example.fishingmod.block;

import net.example.fishingmod.FishingMod;
import net.minecraft.block.Block;
import net.minecraft.block.BlockState;
import net.minecraft.block.entity.BlockEntity;
import net.minecraft.entity.player.PlayerEntity;
import net.minecraft.screen.NamedScreenHandlerFactory;
import net.minecraft.util.ActionResult;
import net.minecraft.util.Hand;
import net.minecraft.util.hit.BlockHitResult;
import net.minecraft.util.math.BlockPos;
import net.minecraft.world.BlockView;
import net.minecraft.world.World;

public class FishBlock extends Block \{
public FishBlock(Settings settings) \{
super(settings);
\}

@Override
public BlockEntity createBlockEntity(BlockView world) \{
return new FishBlockEntity();
\}

@Override
public ActionResult onUse(BlockState state, World world, BlockPos pos,
PlayerEntity player, Hand hand, BlockHitResult hit) \{
if (!world.isClient) \{
NamedScreenHandlerFactory screenHandlerFactory =
(NamedScreenHandlerFactory)world.getBlockEntity(pos);
if (screenHandlerFactory != null) \{
player.openHandledScreen(screenHandlerFactory);
\}
\}
return ActionResult.SUCCESS;
\}
\}

Texture Specifications

custom_fishing_rod.png (16x16)

Color palette: #8B4513 (brown), #D2B48C (light wood), #4169E1 (blue string), #C0C0C0 (metal)

Layout:
- Pole: Diagonal line from (2,2) to (13,14) - brown (#8B4513, 2px thick)
- Handle: (13,12)-(15,15) - dark brown (#654321)
- Reel: Circle at (11,10)-(13,12) - metal (#C0C0C0)
- String: Thin line (14,3)-(15,1) - blue (#4169E1)
- Hook: Small triangle (15,0)-(15,1)-(16,1) - metal (#C0C0C0)

fishing_net.png (16x16) - Idle state

Color palette: #8B4513 (wood), #F4A460 (net), #228B22 (handles)

Layout:
- Wooden frame: Rectangle (2,2)-(14,14) - wood (#8B4513, 1px thick)
- Net mesh: Grid pattern 3x3 squares inside frame - sandy (#F4A460)
- Handles: (1,6)-(2,10) and (14,6)-(15,10) - green (#228B22)

fishing_net_swinging.png (16x16) - Animation state

Color palette: Same as fishing_net.png + motion blur #D4AF37 (yellow streaks)

Layout:
- Same frame as fishing_net.png
- Net mesh: Distorted diagonally (tilted 20° right) - sandy (#F4A460)
- Motion lines: 3 yellow streaks (#D4AF37) from (8,4)-(12,8)
- Handles: Slightly stretched (1,5)-(2,11) and (14,5)-(15,11)

fish_block.png (16x16)

Color palette: #4682B4 (water), #FF6347 (fish), #32CD32 (scales)

Layout:
- Water background: Gradient (0,0)-(15,15) dark to light blue
- Fish body: Oval (3,6)-(13,10) - salmon (#FF6347)
- Fish tail: Triangle (13,7)-(15,9)-(13,11) - darker salmon (#E55A3D)
- Eye: White circle (10,7)-(11,8) with black pupil (10,7)-(11,8)
- Scales: 4 green dots (#32CD32) along body

Texture Generation Methods

  • Online Editors: Use Piskel (piskelapp.com), Lospec Pixel Editor, or Aseprite Online
  • Minecraft Tools: Generate via minecraft.tools/en/texture.php (16x16 canvas)
  • SVG Conversion: Create SVG guides, then convert to PNG at svg-png-converter.com

Verification Steps

  1. Run `gradlew runClient`
  2. Check Creative inventory for proper rendering
  3. Test fishing net right-click for swing animation
  4. Verify fish block texture on placed blocks
